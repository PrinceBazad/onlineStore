import React, { useState } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { useCart } from '../context/CartContext.jsx';
import { useData } from '../context/DataContext.jsx';
import { useAuth } from '../context/AuthContext.jsx';
import { formatINR } from '../utils/format.js';
import { payWithRazorpay } from '../utils/razorpay.js';

const ALL_PAY_METHODS = [
  { id: 'upi', label: 'UPI', hint: 'GPay / PhonePe / Paytm / BHIM' },
  { id: 'card', label: 'Credit / Debit Card', hint: 'Visa, Mastercard, RuPay' },
  { id: 'cod', label: 'Cash on Delivery', hint: 'Pay in cash at your doorstep' },
];

export default function Checkout() {
  const { cart, subtotal, clearCart } = useCart();
  const { placeOrder, settings, products } = useData();
  const { user } = useAuth();
  const nav = useNavigate();

  const [shipping, setShipping] = useState({
    name: user?.name || '',
    phone: user?.phone || '',
    email: user?.email || '',
    address: '',
    city: '',
    state: '',
    pincode: '',
  });
  const [method, setMethod] = useState('');
  const [paying, setPaying] = useState(false);
  const [placed, setPlaced] = useState(null);
  const [err, setErr] = useState('');
  const shippingFee = cart.reduce((sum, c) => {
    const p = products.find((x) => x.id === c.id);
    const perUnit = p && typeof p.shippingCost === 'number' ? p.shippingCost : 99;
    return sum + perUnit * c.qty;
  }, 0);
  const freeThreshold = settings.freeShippingThreshold || 1499;
  const effectiveShipping = subtotal >= freeThreshold ? 0 : shippingFee;
  const total = subtotal + effectiveShipping;

  const availableMethods = ALL_PAY_METHODS.filter((m) =>
    cart.every((c) => {
      const p = products.find((x) => x.id === c.id);
      const accepted = p && Array.isArray(p.paymentMethods) ? p.paymentMethods : ['upi', 'card', 'cod'];
      return accepted.includes(m.id);
    })
  );

  const gatewayConfigured = Boolean(settings.razorpayKeyId);
  const activeMethod = availableMethods.find((m) => m.id === method) ? method : (availableMethods[0]?.id || '');

  if (cart.length === 0 && !placed) {
    return (
      <main className="page center">
        <p>Your cart is empty.</p>
        <Link to="/catalog" className="btn btn-dark">Browse products</Link>
      </main>
    );
  }
  const setShip = (k) => (e) => setShipping({ ...shipping, [k]: e.target.value });

  const validate = () => {
    for (const k of ['name', 'phone', 'address', 'city', 'state', 'pincode']) {
      if (!shipping[k].trim()) return 'Please fill all shipping details.';
    }
    if (!/^\d{6}$/.test(shipping.pincode)) return 'Enter a valid 6-digit pincode.';
    return '';
  };

  const submit = async (e) => {
    e.preventDefault();
    setErr('');
    const v = validate();
    if (v) return setErr(v);

    const baseOrder = {
      userId: user?.id || null,
      items: cart,
      customer: {
        name: shipping.name,
        phone: shipping.phone,
        email: shipping.email,
      },
      shipping: {
        address: shipping.address,
        city: shipping.city,
        state: shipping.state,
        pincode: shipping.pincode,
      },
      subtotal,
      shippingFee: effectiveShipping,
      total,
    };

    if (activeMethod === 'cod') {
      const order = placeOrder({
        ...baseOrder,
        payment: { method: 'cod', status: 'pending', mode: 'Cash on Delivery' },
      });
      clearCart();
      setPlaced(order);
      return;
    }

    if (!gatewayConfigured) {
      setErr(
        'Payment gateway not configured. Ask the admin to add the Razorpay Key ID in Admin → Settings.'
      );
      return;
    }

    setPaying(true);
    try {
      const response = await payWithRazorpay({
        key: settings.razorpayKeyId,
        amount: total,
        name: settings.storeName,
        description: `Order for ${cart.length} item(s)`,
        mode: activeMethod === 'upi' ? 'UPI' : 'Card',
        prefill: {
          name: shipping.name,
          email: shipping.email,
          contact: shipping.phone,
        },
      });

      const order = placeOrder({
        ...baseOrder,
        payment: {
          method: 'online',
          gateway: 'Razorpay',
          status: 'paid',
          mode: activeMethod === 'upi' ? 'UPI' : 'Card',
          ref: response.razorpay_payment_id,
        },
      });
      clearCart();
      setPlaced(order);
    } catch (ex) {
      setErr(ex.message || 'Payment failed. Please try again.');
    } finally {
      setPaying(false);
    }
  };

  if (placed) {
    return (
      <main className="page center">
        <div className="placed-card">
          <span className="placed-tick">✓</span>
          <h1>Order placed!</h1>
          <p>
            Thank you for shopping with us. Your order ID is{' '}
            <strong>{placed.id}</strong>.
          </p>
          <p className="muted">
            {placed.payment.method === 'cod'
              ? 'Pay ₹' + placed.total.toLocaleString('en-IN') + ' on delivery.'
              : `Payment via ${placed.payment.mode} confirmed. Reference: ${placed.payment.ref}`}
          </p>
          <div className="row-gap center m-top">
            <Link to={`/track?order=${placed.id}`} className="btn btn-gold">
              Track order
            </Link>
            <Link to="/catalog" className="btn btn-ghost">
              Continue shopping
            </Link>
          </div>
        </div>
      </main>
    );
  }

  return (
    <main className="page checkout">
      <div className="page-head">
        <h1>Checkout</h1>
        <p>Complete your order</p>
      </div>

      <form onSubmit={submit}>
        <div className="co-left">
          <section className="card-box">
            <h2>1 · Shipping details</h2>
            <div className="grid2">
              <label>
                Full name
                <input value={shipping.name} onChange={setShip('name')} />
              </label>
              <label>
                Phone
                <input value={shipping.phone} onChange={setShip('phone')} />
              </label>
            </div>
            <label>
              Email
              <input type="email" value={shipping.email} onChange={setShip('email')} />
            </label>
            <label>
              Address
              <input value={shipping.address} onChange={setShip('address')} placeholder="House no, street, area" />
            </label>
            <div className="grid3">
              <label>
                City
                <input value={shipping.city} onChange={setShip('city')} />
              </label>
              <label>
                State
                <input value={shipping.state} onChange={setShip('state')} />
              </label>
              <label>
                Pincode
                <input value={shipping.pincode} onChange={setShip('pincode')} maxLength={6} />
              </label>
            </div>
          </section>
      shippingFee: effectiveShipping,
      total,
    };

    if (activeMethod === 'cod') {
      const payment = { method: 'cod', status: 'pending', mode: 'Cash on Delivery' };
      setPlaced({ ...baseOrder, payment });
      placeOrder({ ...baseOrder, payment });
      clearCart();
      return;
    }

    if (!gatewayConfigured) {
      setErr('Payment gateway not configured. Ask the admin to add the Razorpay Key ID in Admin \u2192 Settings.');
      return;
    }

    setPaying(true);
    try {
      const response = await payWithRazorpay({
        key: settings.razorpayKeyId,
        amount: total,
        name: settings.storeName,
        mode: activeMethod === 'upi' ? 'UPI' : 'Card',
        prefill: { name: shipping.name, email: shipping.email, contact: shipping.phone },
      });
      const payment = {
        method: 'online',
        gateway: 'Razorpay',
        status: 'paid',
        mode: activeMethod === 'upi' ? 'UPI' : 'Card',
        ref: response.razorpay_payment_id,
      };
      setPlaced({ ...baseOrder, payment });
      placeOrder({ ...baseOrder, payment });
      clearCart();
    } catch (ex) {
      if (ex.code === 'BAD_REQUEST_ERROR' || String(ex.message).includes('400')) {
        await runDemoPayment(total, activeMethod, baseOrder);
      } else {
        setPaying(false);
        setErr(ex.message || 'Payment failed. Please try again.');
      }
    }
  };  if (placed) {
    return (
      <main className="page center">
        <div className="card-box" style={{ maxWidth: 560, margin: '0 auto', textAlign: 'center' }}>
          <div style={{ fontSize: '3rem', marginBottom: 12 }}>\u2705</div>
          <h1>Order placed!</h1>
          <p className="muted">Your order <strong>{placed.payment.ref || 'ORD-' + Date.now().toString(36).toUpperCase()}</strong> has been confirmed.</p>
          <p className="muted">Payment: {placed.payment.mode} ({placed.payment.gateway || 'COD'})</p>
          <h3 style={{ marginTop: 18 }}>Order total: {formatINR(placed.total)}</h3>
          <div className="row-gap" style={{ justifyContent: 'center', marginTop: 18 }}>
            <Link to="/" className="btn btn-dark">Back to home</Link>
            <Link to="/orders" className="btn btn-gold">View my orders</Link>
          </div>
        </div>
      </main>
    );
  }

  return (
    <main className="page checkout">
      <DemoPaymentModal
        open={demoModal.open}
        amount={demoModal.amount}
        mode={demoModal.mode}
        onConfirm={demoModal.onConfirm}
        onCancel={() => { setDemoModal({ open: false, amount: 0, mode: 'upi', onConfirm: null }); setPaying(false); }}
      />
      <div className="page-head">
        <h1>Checkout</h1>
        <p>Complete your order</p>
      </div>
      <form onSubmit={submit} className="checkout-grid">
        <div className="co-left">
          <section className="card-box">
            <h2>1 \u00b7 Shipping details</h2>
            <div className="grid2">
              <label>Full name<input value={shipping.name} onChange={setShip('name')} /></label>
              <label>Phone<input value={shipping.phone} onChange={setShip('phone')} /></label>
            </div>
            <label>Email<input type="email" value={shipping.email} onChange={setShip('email')} /></label>
            <label>Address<input value={shipping.address} onChange={setShip('address')} placeholder="House no, street, area" /></label>
            <div className="grid3">
              <label>City<input value={shipping.city} onChange={setShip('city')} /></label>
              <label>State<input value={shipping.state} onChange={setShip('state')} /></label>
              <label>Pincode<input value={shipping.pincode} onChange={setShip('pincode')} maxLength={6} /></label>
            </div>
          </section>          <section className="card-box">
            <h2>2 \u00b7 Payment method</h2>
            <div className="pay-methods">
              {availableMethods.map((m) => (
                <label key={m.id} className={`pay-opt ${activeMethod === m.id ? 'active' : ''}`}>
                  <input type="radio" name="pay" checked={activeMethod === m.id} onChange={() => setMethod(m.id)} />
                  <div>
                    <strong>{m.label}</strong>
                    <span>{m.hint}</span>
                  </div>
                </label>
              ))}
            </div>
            {activeMethod === 'upi' && (
              <div className="pay-fields">
                <p className="muted">You\u2019ll finish paying with your UPI app inside the Razorpay window (GPay / PhonePe / Paytm / BHIM).</p>
              </div>
            )}
            {activeMethod === 'card' && (
              <div className="pay-fields">
                <p className="muted">You\u2019ll enter your card details inside the Razorpay window (Visa / Mastercard / RuPay).</p>
              </div>
            )}
            {activeMethod === 'cod' && (
              <p className="muted">\u2705 Please keep exact change ready. Our delivery partner will collect payment on arrival.</p>
            )}
            {activeMethod !== 'cod' && !gatewayConfigured && (
              <p className="error">\u26a0 Payment gateway not configured \u2014 the admin needs to add the Razorpay Key ID in Admin \u2192 Settings.</p>
            )}
            {activeMethod !== 'cod' && gatewayConfigured && (
              <p className="ok tiny">\u2713 Secure online payment via Razorpay. Your card / UPI details are never stored on this site.</p>
            )}
          </section>
        </div>

        <aside className="co-right">
          <div className="card-box sticky">
            <h2>Order summary</h2>
            <ul className="sum-items">
              {cart.map((c) => (
                <li key={c.id}>
                  <span>{c.name} \u00d7 {c.qty}</span>
                  <span>{formatINR(c.price * c.qty)}</span>
                </li>
              ))}
            </ul>
            <div className="sum-lines">
              <div><span>Subtotal</span><span>{formatINR(subtotal)}</span></div>
              <div><span>Shipping</span><span>{effectiveShipping === 0 ? 'FREE' : formatINR(effectiveShipping)}</span></div>
              <div className="total"><span>Total</span><span>{formatINR(total)}</span></div>
            </div>
            {err && <p className="error">{err}</p>}
            <button className="btn btn-gold btn-block" type="submit" disabled={paying}>
              {paying ? 'Processing payment\u2026' : activeMethod === 'cod' ? 'Place order \u00b7 Pay on delivery' : 'Pay & place order'}
            </button>
            <p className="muted tiny">{activeMethod === 'cod' ? '\uD83D\uDD12 No advance payment needed.' : '\uD83D\uDD12 Your payment details are encrypted and secure.'}</p>
          </div>
        </aside>
      </form>
    </main>
  );
}
